# Become the Meme — dependencies

# --- Core ---
numpy
Pillow

# --- Webcam capture + image ops (platform-agnostic) ---
# Use ONLY opencv-contrib-python (mediapipe depends on it). Installing plain
# opencv-python alongside it corrupts the shared cv2/ dir and breaks the GUI
# (windows that display but ignore keyboard/close events).
opencv-contrib-python

# --- Person segmentation & body-pose keypoints (local, no cloud) ---
mediapipe

# --- Zero-shot image embeddings via CLIP (local, no training data needed) ---
# torch provides Apple Silicon (MPS) / CUDA / CPU backends automatically.
torch
open_clip_torch

# --- Text encoders ---
# transformers is REQUIRED by the default (concept) backend: SigLIP2's tokenizer
# loads through Hugging Face transformers. It also powers the vlm backend (Qwen2-VL).
# sentence-transformers is used by the vlm backend for text matching.
transformers
sentence-transformers
